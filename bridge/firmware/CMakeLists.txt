cmake_minimum_required(VERSION 3.20)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH to the pinned external SDK; no download is performed here.")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(simplecontroller_bridge C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(NOT PICO_BOARD STREQUAL "pico2_w" AND NOT PICO_BOARD STREQUAL "pico_w")
  message(FATAL_ERROR "This prototype supports only an explicitly selected pico2_w or pico_w target.")
endif()
if(NOT DEFINED SC_USB_VID OR NOT DEFINED SC_USB_PID)
  message(FATAL_ERROR "Provide SC_USB_VID and SC_USB_PID explicitly. No assigned product identity is claimed.")
endif()
set(SC_USB_POWER_MA 500 CACHE STRING "Advertised USB budget; physical supply/current still must be verified")
option(SC_TRIGGER_MCP4251 "Enable only with the documented supervised analog trigger circuit" OFF)
math(EXPR SC_USB_POWER_REMAINDER "${SC_USB_POWER_MA} % 2")
if(SC_USB_POWER_MA LESS 2 OR SC_USB_POWER_MA GREATER 500 OR SC_USB_POWER_REMAINDER)
  message(FATAL_ERROR "SC_USB_POWER_MA must be an even number from 2 through 500; lowering it does not establish low-power operation.")
endif()
set(SC_BRIDGE_CONFIG "${CMAKE_CURRENT_LIST_DIR}/config.example.h" CACHE FILEPATH "Private header with local Wi-Fi configuration")
configure_file("${SC_BRIDGE_CONFIG}" "${CMAKE_CURRENT_BINARY_DIR}/private-config/bridge_config.h" COPYONLY)
add_executable(simplecontroller_bridge main.cpp usb_descriptors.cpp ../src/scx2.cpp)
target_include_directories(simplecontroller_bridge PRIVATE . ../include "${CMAKE_CURRENT_BINARY_DIR}/private-config")
# TinyUSB's C-compatible headers intentionally use GNU extensions. Keep strict
# warnings on our C++17 sources without diagnosing upstream headers as ours.
target_include_directories(simplecontroller_bridge SYSTEM PRIVATE "${PICO_TINYUSB_PATH}/src")
target_compile_definitions(simplecontroller_bridge PRIVATE SC_USB_VID=${SC_USB_VID} SC_USB_PID=${SC_USB_PID} SC_USB_POWER_MA=${SC_USB_POWER_MA})
target_compile_definitions(simplecontroller_bridge PRIVATE SC_TRIGGER_MCP4251=$<BOOL:${SC_TRIGGER_MCP4251}>)
target_compile_options(simplecontroller_bridge PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>)
target_link_libraries(simplecontroller_bridge PRIVATE pico_stdlib pico_rand hardware_watchdog hardware_spi pico_cyw43_arch_lwip_poll tinyusb_device)
pico_enable_stdio_usb(simplecontroller_bridge 0)
pico_enable_stdio_uart(simplecontroller_bridge 0)
pico_set_program_name(simplecontroller_bridge "SimpleController SCX2 experimental bridge")
pico_set_program_version(simplecontroller_bridge "0.1.0")
pico_add_extra_outputs(simplecontroller_bridge)
